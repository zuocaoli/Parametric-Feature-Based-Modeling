cmake_minimum_required(VERSION 4.3)
set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
project(mini_gcs_month01_week01 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_MODULE_STD ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 22)
    message(FATAL_ERROR "Use Clang 22+ and matching clang-scan-deps / clangd.")
endif()
if(NOT 23 IN_LIST CMAKE_CXX_COMPILER_IMPORT_STD)
    message(FATAL_ERROR "Install libstdc++ 16 module development files for import std.")
endif()
option(MINI_GCS_BUILD_GUI "Build the Qt Widgets observer" ON)
include(CTest)
add_library(mini_gcs STATIC src/model.cpp src/lesson.cpp)
target_sources(mini_gcs PUBLIC FILE_SET CXX_MODULES FILES src/model.cppm src/lesson.cppm)
target_compile_options(mini_gcs PRIVATE -Wall -Wextra -Wpedantic)
if(MINI_GCS_BUILD_GUI)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets)
    add_library(mini_gcs_ui STATIC ui/workbench.cpp)
    target_include_directories(mini_gcs_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(mini_gcs_ui PUBLIC mini_gcs Qt6::Widgets)
    target_compile_options(mini_gcs_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(demo main.cpp)
if(MINI_GCS_BUILD_GUI)
    target_compile_definitions(demo PRIVATE MINI_GCS_WITH_GUI=1)
    target_link_libraries(demo PRIVATE mini_gcs_ui)
else()
    target_link_libraries(demo PRIVATE mini_gcs)
endif()
add_executable(console-demo main.cpp)
target_link_libraries(console-demo PRIVATE mini_gcs)
if(BUILD_TESTING)
    add_executable(core-tests tests/core_tests.cpp)
    target_link_libraries(core-tests PRIVATE mini_gcs)
    foreach(scenario model)
        add_test(NAME ${scenario} COMMAND core-tests ${scenario})
    endforeach()
    add_test(NAME console COMMAND console-demo --headless)
    if(MINI_GCS_BUILD_GUI)
        find_package(Qt6 6.11 REQUIRED COMPONENTS Test)
        add_executable(ui-tests tests/ui_tests.cpp)
        target_link_libraries(ui-tests PRIVATE mini_gcs_ui Qt6::Test)
        add_test(NAME ui COMMAND ui-tests)
        set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
