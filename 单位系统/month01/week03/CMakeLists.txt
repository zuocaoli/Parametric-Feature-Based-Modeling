cmake_minimum_required(VERSION 4.3)
set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
project(units_month01_week03 LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 22)
    message(FATAL_ERROR "Use Clang/clang-scan-deps 22 and the default preset.")
endif()
if(NOT 23 IN_LIST CMAKE_CXX_COMPILER_IMPORT_STD)
    message(FATAL_ERROR "Install libstdc++ 16 standard-library module development files.")
endif()
set(CMAKE_CXX_MODULE_STD ON)
add_compile_options(-Wall -Wextra -Wpedantic)
include(CTest)
add_library(course STATIC document.cpp expression.cpp input.cpp property.cpp quantity.cpp schema.cpp unit.cpp)
target_sources(course PUBLIC FILE_SET CXX_MODULES FILES document.cppm expression.cppm input.cppm property.cppm quantity.cppm schema.cppm unit.cppm)
add_executable(demo main.cpp)
target_link_libraries(demo PRIVATE course)
if(BUILD_TESTING)
    add_executable(document_tests document_tests.cpp)
    target_link_libraries(document_tests PRIVATE course)
    add_test(NAME document_tests COMMAND document_tests)
    add_executable(expression_tests expression_tests.cpp)
    target_link_libraries(expression_tests PRIVATE course)
    add_test(NAME expression_tests COMMAND expression_tests)
    add_executable(input_tests input_tests.cpp)
    target_link_libraries(input_tests PRIVATE course)
    add_test(NAME input_tests COMMAND input_tests)
    add_executable(property_tests property_tests.cpp)
    target_link_libraries(property_tests PRIVATE course)
    add_test(NAME property_tests COMMAND property_tests)
    add_executable(quantity_tests quantity_tests.cpp)
    target_link_libraries(quantity_tests PRIVATE course)
    add_test(NAME quantity_tests COMMAND quantity_tests)
    add_executable(schema_tests schema_tests.cpp)
    target_link_libraries(schema_tests PRIVATE course)
    add_test(NAME schema_tests COMMAND schema_tests)
    add_test(NAME demo_acceptance COMMAND demo)
endif()
